    function normalizeText(str) {

      return str

        .toLowerCase()

        .replace(/[.,\-_/()]/g, " ")

        .replace(/\s+/g, " ")

        .trim();

    }

    function matchCityTokens(item, qNorm, tokens) {

      if (tokens.length === 0) return true;

      const targetNorm = normalizeText(`${item.k} ${item.p}`);

      const targetCompact = targetNorm.replace(/\s+/g, "");

      const qCompact = qNorm.replace(/\s+/g, "");

      const allTokensMatch = tokens.every(tok => targetNorm.includes(tok));

      const compactMatch = targetCompact.includes(qCompact);

      return allTokensMatch || compactMatch;

    }
